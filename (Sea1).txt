local function stopCurrentTween()
    if currentTween then currentTween:Cancel(); currentTween:Destroy(); currentTween = nil; end
end

local function tweenTo(rootPart, targetPos)
    local distance = (rootPart.Position - targetPos).Magnitude
    if distance < 3 then stopCurrentTween(); return end 

    if currentTween and currentTween.PlaybackState == Enum.PlaybackState.Playing then
        if (targetPos - lastTargetPos).Magnitude < 3 then return end
    end

    stopCurrentTween() 
    lastTargetPos = targetPos

    local duration = distance / TWEEN_SPEED
    local tweenInfo = TweenInfo.new(duration, Enum.EasingStyle.Linear, Enum.EasingDirection.Out)
    
    pcall(function()
        rootPart.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
        rootPart.AssemblyAngularVelocity = Vector3.new(0, 0, 0)
    end)
    
    currentTween = TweenService:Create(rootPart, tweenInfo, {CFrame = CFrame.new(targetPos) * (rootPart.CFrame - rootPart.Position)})
    currentTween:Play()
end

local function safeTeleportToGate(rootPart, gatePos3D, gateVector)
    stopCurrentTween(); task.wait(1);
    if CommF then pcall(function() CommF:InvokeServer("requestEntrance", gateVector) end) end
    task.wait(0.2)
    rootPart.CFrame = CFrame.new(gatePos3D)
    task.wait(0.4) 
end

-- Kiểm tra UI Quest Kép (Quest Title + Guide Subtitle)
local function hasActiveQuest()
    local playerGui = localPlayer:FindFirstChildOfClass("PlayerGui")
    local mainGui = playerGui and playerGui:FindFirstChild("Main")
    if not mainGui then return false end

    -- Check 1: Main -> Quest -> Container -> QuestTitle -> Title
    local questFrame = mainGui:FindFirstChild("Quest")
    if questFrame and questFrame.Visible then
        local container = questFrame:FindFirstChild("Container")
        local questTitle = container and container:FindFirstChild("QuestTitle")
        local titleLabel = questTitle and questTitle:FindFirstChild("Title")
        if titleLabel and titleLabel.Text and titleLabel.Text ~= "" and string.find(titleLabel.Text, "Defeat") then
            return true
        end
    end

    -- Check 2: Main -> Guide -> TopFrame -> Subtitle1
    local guideFrame = mainGui:FindFirstChild("Guide")
    if guideFrame then
        local topFrame = guideFrame:FindFirstChild("TopFrame")
        local subtitle1 = topFrame and topFrame:FindFirstChild("Subtitle1")
        if subtitle1 and subtitle1.Text and subtitle1.Text ~= "" and string.find(subtitle1.Text, "Defeat") then
            return true
        end
    end

    return false
end

local function ApplyNoclipFullEnemy(enemy)
    if enemy ~= nil then
        for _, child in pairs(enemy:GetDescendants()) do
            if child:IsA("BasePart") and child.CanCollide == true then child.CanCollide = false end
        end
    end
end

local function PrepareMobState(enemy, hrp, humanoid)
    if enemy:GetAttribute("BusoEnabled") ~= nil then enemy:SetAttribute("BusoEnabled", false) end
    enemy:SetAttribute("Clashable", true)

    if humanoid then
        if not humanoid.Sit then humanoid.Sit = false end
        if not humanoid.PlatformStand then humanoid.PlatformStand = false end
    end

    pcall(function()
        if hrp.Anchored then hrp.Anchored = false end
        if hrp.SetNetworkOwner then hrp:SetNetworkOwner(localPlayer) end
    end)
    ApplyNoclipFullEnemy(enemy)
end

-- Module Bring Mobs Tụ Tại Map (Khóa Lực Trục Y)
task.spawn(function()
    while true do
        local _, _, questMobName, defaultTargetPos, _ = getCurrentIslandConfig()
        
        if defaultTargetPos and questMobName and questMobName ~= "" and hasActiveQuest() then
            local targetPos = defaultTargetPos
            
            for _, enemy in pairs(enemiesFolder:GetChildren()) do
                if enemy.Name == questMobName then 
                    local hrp = enemy:FindFirstChild("HumanoidRootPart")
                    local humanoid = enemy:FindFirstChildOfClass("Humanoid")
                    
                    if hrp then
                        PrepareMobState(enemy, hrp, humanoid)
                        local bp = hrp:FindFirstChild("BringMob")
                        local distance = (hrp.Position - targetPos).Magnitude
                        
                        if bp then
                            bp.MaxForce = Vector3.new(100000, 0, 100000)
                            bp.Position = targetPos
                            
                            if distance <= MAX_DISTANCE_W then
                                bp.P = 1550; bp.D = 425;
                            elseif distance <= MAX_DISTANCE then
                                bp.P = 3000; bp.D = 225;
                            end
                        else
                            if distance <= MAX_DISTANCE then
                                pcall(function() hrp.CFrame = CFrame.new(targetPos) end)
                                
                                local newBp = Instance.new("BodyPosition")
                                newBp.Name = "BringMob"
                                newBp.MaxForce = Vector3.new(100000, 0, 100000)
                                newBp.Position = targetPos
                                newBp.Parent = hrp
                            end
                        end
                    end
                end
            end
        end
        task.wait(0.1)
    end
end)

enemiesFolder.ChildAdded:Connect(function(newEnemy)
    task.wait(0.05)
    local _, _, questMobName, defaultTargetPos, _ = getCurrentIslandConfig()

    if defaultTargetPos and hasActiveQuest() and questMobName and newEnemy.Name == questMobName then
        local targetPos = defaultTargetPos
        local hrp = newEnemy:FindFirstChild("HumanoidRootPart")
        if hrp then
            local distance = (hrp.Position - targetPos).Magnitude
            if distance <= MAX_DISTANCE then
                local humanoid = newEnemy:FindFirstChildOfClass("Humanoid")
                PrepareMobState(newEnemy, hrp, humanoid)
            end
        end
    end
end)

-- Vòng lặp chính: Điều hướng & Auto Quest
task.spawn(function()
    while task.wait(0.15) do 
        local Character = localPlayer.Character
        local RootPart = Character and Character:FindFirstChild("HumanoidRootPart")
        if not RootPart then continue end 

        local currentQuestName, currentQuestNum, mobName, defaultTargetPos, gateVector = getCurrentIslandConfig()
        if not currentQuestName or currentQuestName == "" then continue end

        if not hasActiveQuest() then
            stopCurrentTween()
            if CommF then pcall(function() CommF:InvokeServer("StartQuest", currentQuestName, currentQuestNum) end) end
            task.wait(0.3)
            continue
        end

        local viTriDich = Vector3.new(defaultTargetPos.X, defaultTargetPos.Y + Player_Y, defaultTargetPos.Z)

        if gateVector == FV3 and RootPart.Position.X < 50000 then
            stopCurrentTween()
            if CommF then pcall(function() CommF:InvokeServer("requestEntrance", FV3) end) end
            task.wait(0.8); continue;
        end

        if gateVector then
            local gatePos3D = toVector3(gateVector)
            local distance_HRP_to_Map = (RootPart.Position - viTriDich).Magnitude
            local distance_Gate_to_Map = (gatePos3D - viTriDich).Magnitude
            local distance_HRP_to_Gate = (RootPart.Position - gatePos3D).Magnitude

            if distance_HRP_to_Gate <= 300 then
                if distance_HRP_to_Map > 4 then tweenTo(RootPart, viTriDich) else stopCurrentTween() end
            elseif distance_Gate_to_Map < distance_HRP_to_Map then
                safeTeleportToGate(RootPart, gatePos3D, gateVector)
            else
                if distance_HRP_to_Map > 4 then tweenTo(RootPart, viTriDich) else stopCurrentTween() end
            end
        else
            if (RootPart.Position - viTriDich).Magnitude > 4 then tweenTo(RootPart, viTriDich) else stopCurrentTween() end
        end
    end
end)
